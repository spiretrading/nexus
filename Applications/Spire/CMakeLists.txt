macro(find_cpp_qobject output)
  foreach(_i ${ARGN})
    if(EXISTS "${_i}")
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_i}")
      file(READ "${_i}" _contents)
      string(REGEX MATCHALL "Q_OBJECT" _match "${_contents}")
      if(_match)
        list(APPEND ${output} "${_i}")
      endif()
    endif()
  endforeach()
endmacro()
